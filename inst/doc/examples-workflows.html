Running an example

Each heading below names an installed script. To inspect and run one:

file.show(script)
source(script)

Replace the filename to run another script. The listings below show the complete code without executing it during documentation builds. Sourcing a script runs its fits and checks; resampling, tuning and composition matrices can take longer than the introductory guides.

Training binary and shared multiclass workflows

calibrated-workflow.R: Retain local preprocessing, models and calibration together, then inspect predictions and learning diagnostics.

# Installed example: optional integration; failures propagate when available.
source(system.file("examples", "assertions.R", package = "vennabers"))
packages <- c("parsnip", "workflows", "recipes", "rsample", "hardhat", "generics",
              "rlang", "tibble", "tidyselect", "withr")
if (all(vapply(packages, requireNamespace, logical(1), quietly = TRUE))) local({
  library(vennabers)
  # Development data are available for training and calibration. Assessment
  # observations below are untouched by either learning step or parameter choice.
  set.seed(12)
  data <- data.frame(x = stats::rnorm(250))
  data$truth <- factor(ifelse(stats::runif(250) < stats::plogis(data$x), "yes", "no"),
                       levels = c("yes", "no"))
  development <- data[1:200, ]
  assessment <- data[201:250, ]
  recipe <- recipes::recipe(truth ~ x, development) |>
    recipes::step_impute_median(recipes::all_numeric_predictors()) |>
    recipes::step_normalize(recipes::all_numeric_predictors())
  base <- workflows::workflow() |>
    workflows::add_recipe(recipe) |>
    workflows::add_model(parsnip::logistic_reg(engine = "glm"))
  calibrated <- calibrated_workflow(base, outcome = "truth", seed = 42)
  fitted <- generics::fit(calibrated, development)
  # Prediction needs predictors only; assessment truth is used later for scoring.
  new_data <- assessment["x"]
  probability <- predict(fitted, new_data, type = "prob")
  class <- predict(fitted, new_data, type = "class")
  details <- venn_abers_details(fitted, new_data)
  state <- workflows::extract_fit_engine(fitted)
  assert_conditions(identical(probability, details$prediction), nrow(class) == 50L,
            length(state$models) == 1L, state$event_class == "yes",
            state$folds$Inductive1$training_size + state$folds$Inductive1$calibration_size == 200L)
  # Prediction-time loss changes neither the fitted model nor its stored default.
  brier <- predict(fitted, new_data, type = "prob", loss = "brier")
  stopifnot(identical(brier, venn_abers_details(fitted, new_data, loss = "brier")$prediction),
            workflows::extract_fit_engine(fitted)$loss == "log")
  print(head(probability))
  print(head(details$details))
  print(brier_score(assessment$truth == "yes", probability$.pred_yes))
  # Local recipe statistics are inspectable through the full stored workflow.
  print(workflows::extract_recipe(state$models$Inductive1))
  # This split cannot undo earlier tuning or transformations that used calibration
  # or assessment observations. IID data and a model fixed independently of its
  # calibration observations remain assumptions; VA bounds are not confidence
  # intervals and calibration need not improve scores on a particular test set.
})

  # plan does not establish parity with Python's separate pair-model training.
  data <- iris
  names(data)[5L] <- "truth"
  split <- withr::with_seed(33, rsample::initial_split(data, .8, strata = truth))
  development <- rsample::analysis(split)
  assessment <- rsample::assessment(split)
  base <- workflows::workflow() |>
    workflows::add_recipe(recipes::recipe(truth ~ ., development) |>
      recipes::step_normalize(recipes::all_numeric_predictors())) |>
    workflows::add_model(parsnip::multinom_reg(engine = "nnet"))
  for (classification in c("ovo", "ovr")) {
    fitted <- generics::fit(vennabers::calibrated_workflow(base, "truth", method = "cross",
      classification = classification, seed = 33, precision = 2, loss = "brier"), development)
    p <- predict(fitted, assessment, type = "prob")
    cls <- predict(fitted, assessment, type = "class")
    details <- vennabers::venn_abers_details(fitted, assessment)
    state <- workflows::extract_fit_engine(fitted)
    stopifnot(identical(p, details$prediction), nrow(cls) == nrow(assessment),
              length(state$models) == 5L, setequal(details$details$fold, names(state$models)))
    # Each diagnostic row is a binary fold result: aggregate across folds first,
    # then couple OvO pairs or normalize OvR classes. No multiclass intervals.
    print(head(details$details))
    matrix_p <- as.matrix(p)
    colnames(matrix_p) <- state$class_levels
    print(c(brier = vennabers::brier_score(assessment$truth, matrix_p),
            log_loss = vennabers::log_loss(assessment$truth, matrix_p)))
  }
})

Encoding binary names

encoded-binary.R: Preserve reserved predictor and class names through encoding, native resampling and decoded predictions.

source(system.file("examples", "check-encoded-evaluation.R", package = "vennabers"), local = TRUE)

if (all(vapply(c("parsnip", "workflows", "hardhat", "rsample", "recipes", "rlang", "tibble",
                 "withr", "generics", "tidyselect", "tune", "yardstick"),
               requireNamespace, logical(1), quietly = TRUE))) {
  development <- data.frame(x = sin(1:80), y = factor(rep(c("class", ".pred_yes", ".pred_yes", "class"), 20),
                                                    levels = c("class", ".pred_yes")))
  names(development) <- c("vaV0001", ".pred_class")
  schema <- vennabers::venn_abers_schema(development, outcome = ".pred_class", event_level = "second")
  encoded <- vennabers::venn_abers_encode(development, schema)
  # Build formulas/recipes explicitly in the encoded namespace.
  formula <- reformulate(setdiff(names(encoded), schema$encoded_outcome), schema$encoded_outcome)
  base <- workflows::workflow() |>
    workflows::add_recipe(recipes::recipe(formula, encoded) |>
                            recipes::step_normalize(recipes::all_numeric_predictors())) |>
    workflows::add_model(parsnip::logistic_reg(engine = "glm"))
  model <- vennabers::calibrated_workflow(base, schema$encoded_outcome, schema = schema, seed = 42)
  fitted <- generics::fit(model, encoded)
  new_data <- vennabers::venn_abers_encode(development[1:3, "vaV0001", drop = FALSE], schema)
  probability <- vennabers::venn_abers_decode(predict(fitted, new_data, type = "prob"), schema)$probability
  class <- vennabers::venn_abers_decode(predict(fitted, new_data, type = "class"), schema)$class
  details <- vennabers::venn_abers_decode(vennabers::venn_abers_details(fitted, new_data), schema)$details
  stopifnot(identical(colnames(probability), c("class", ".pred_yes")), length(class) == 3L)

  # Preserve prescribed memberships and repeat/fold IDs when rebuilding on encoded data.
  original_outer <- withr::with_seed(41, rsample::vfold_cv(development, v = 2, repeats = 2))
  outer <- rsample::manual_rset(lapply(original_outer$splits, function(s) {
    rsample::make_splits(list(analysis = as.integer(s, data = "analysis"),
                             assessment = as.integer(s, data = "assessment")), encoded)
  }), original_outer[c("id", "id2")])
  control <- tune::control_resamples(save_pred = TRUE, extract = workflows::extract_fit_engine,
    event_level = schema$event_level, parallel_over = "resamples", allow_par = FALSE, pkgs = c("vennabers", "stats"))
  evaluation <- withr::with_seed(43, tune::fit_resamples(model, outer,
    metrics = yardstick::metric_set(yardstick::brier_class, yardstick::mn_log_loss), control = control))
  checked <- check_encoded_evaluation(evaluation, outer, schema)
  # Truth and original outcome never overwrite .pred_class or the outer IDs.
  stopifnot(identical(checked$decoded$outcome, ".pred_class"), nrow(checked$decoded$identity) == 160L,
            all(vapply(tune::collect_extracts(evaluation)$.extracts, function(s) identical(s$schema, schema), logical(1))))
}

Encoding multiclass and regression names

encoded-multiclass-regression.R: Use explicit schemas for shared OvO/OvR and regression with Inductive/Cross calibration and native evaluation.

# Installed examples: shared OvO/OvR and regression, Inductive and Cross.
source(system.file("examples", "check-encoded-evaluation.R", package = "vennabers"), local = TRUE)
if (all(vapply(c("parsnip", "workflows", "hardhat", "rsample", "recipes", "rlang", "tibble",
                 "withr", "generics", "tidyselect", "tune", "yardstick", "nnet"),
               requireNamespace, logical(1), quietly = TRUE))) {
  examples <- list()
  for (classification in c("ovo", "ovr", "regression")) {
    regression <- classification == "regression"
    development <- data.frame(x = sin(1:120), y = if (regression) 3 + sin(1:120) + cos(1:120) else
      factor(rep(c("class", ".pred_yes", "ümlaut"), 40), levels = c("ümlaut", "class", ".pred_yes")))
    names(development) <- c("vaV0001", if (regression) ".pred" else ".pred_class")
    schema <- if (regression) vennabers::venn_abers_schema(development, ".pred") else
      vennabers::venn_abers_schema(development, ".pred_class", classification = classification)
    encoded <- vennabers::venn_abers_encode(development, schema)
    formula <- reformulate(names(encoded)[1], schema$encoded_outcome)
    base <- workflows::workflow() |>
      workflows::add_recipe(recipes::recipe(formula, encoded) |>
                              recipes::step_normalize(recipes::all_numeric_predictors())) |>
      workflows::add_model(if (regression) parsnip::linear_reg(engine = "lm") else
        parsnip::multinom_reg(engine = "nnet", penalty = .01))
    # Rebuild prescribed outer memberships and IDs on the encoded data.
    original_outer <- withr::with_seed(41, rsample::vfold_cv(development, v = 2, repeats = 2))
    outer <- rsample::manual_rset(lapply(original_outer$splits, function(s) {
      rsample::make_splits(list(analysis = as.integer(s, data = "analysis"),
                               assessment = as.integer(s, data = "assessment")), encoded)
    }), original_outer[c("id", "id2")])
    scoring <- if (regression) yardstick::metric_set(yardstick::rmse, yardstick::mae) else
      yardstick::metric_set(yardstick::brier_class, yardstick::mn_log_loss)
    control <- tune::control_resamples(save_pred = TRUE, extract = identity, allow_par = FALSE,
      parallel_over = "resamples", pkgs = c("vennabers", "stats", "nnet"))
    for (method in c("inductive", "cross")) {
      model <- vennabers::calibrated_workflow(base, schema$encoded_outcome, schema = schema,
                                             method = method, seed = 42)
      fitted <- generics::fit(model, encoded)
      new <- vennabers::venn_abers_encode(development[1:3, "vaV0001", drop = FALSE], schema)
      native <- predict(fitted, new, type = if (regression) "numeric" else "prob")
      decoded <- vennabers::venn_abers_decode(native, schema)
      details <- vennabers::venn_abers_decode(vennabers::venn_abers_details(fitted, new), schema)
      if (!regression) {
        class <- vennabers::venn_abers_decode(predict(fitted, new, type = "class"), schema)$class
        stopifnot(identical(colnames(decoded$probability), schema$class_levels), identical(levels(class), schema$class_levels))
      } else stopifnot(is.double(decoded$prediction), is.null(decoded$probability), is.null(decoded$class))
      evaluation <- withr::with_seed(43, tune::fit_resamples(model, outer, metrics = scoring, control = control))
      checked <- check_encoded_evaluation(evaluation, outer, schema)
      stopifnot(nrow(checked$decoded$identity) == 2L * nrow(development), is.null(checked$decoded$details))
      examples[[paste(classification, method)]] <- list(schema = schema, data = encoded, base = base,
        model = model, fit = fitted, outer = outer, evaluation = evaluation, decoded = decoded, details = details)
    }
  }
}

Reference-style data roles

reference-splits.R: Illustrate ordered holdouts and explicit fold complements on synthetic data, without claiming paper benchmark reproduction.

# Reference-style row partitions; no downloads or new splitter API.
# Inspired by the binary, multiclass and regression applications of the
# Venn-Abers papers. These small synthetic examples illustrate roles, not
# reproductions of paper benchmarks.
library(vennabers)
source(system.file("examples", "assertions.R", package = "vennabers"))

# Calibration gets ceiling(n * calibration_prop) rows; training gets the
# preceding n - ceiling(n * calibration_prop). Never sort, shuffle or stratify.
ordered_holdout <- function(data, calibration_prop = .25) {
  assert_conditions(is.numeric(calibration_prop), length(calibration_prop) == 1L,
            is.finite(calibration_prop), calibration_prop > 0, calibration_prop < 1)
  n <- nrow(data)
  cut <- n - ceiling(n * calibration_prop)
  stopifnot(cut >= 1L, cut < n)
  rsample::make_splits(list(analysis = seq_len(cut),
                           assessment = seq.int(cut + 1L, n)), data)
}

if (all(vapply(c("parsnip", "workflows", "rsample", "recipes", "nnet", "withr"),
               requireNamespace, logical(1), quietly = TRUE))) {
  # A small known membership, including rounding of 11 * .25 to 3 held rows.
  small <- data.frame(x = 11:1)
  stopifnot(identical(as.integer(ordered_holdout(small), data = "analysis"), 1:8),
            identical(as.integer(ordered_holdout(small), data = "assessment"), 9:11))

  # Non-shuffled, stratified three-fold reference assignment. Class order is
  # first appearance (b, a, c); class counts are 3, 4, 4. Remainders are spread
  # across folds (sizes 4, 4, 3). Positions within each class keep their order.
  small$truth <- factor(c("b", "b", "a", "a", "a", "c", "c", "c", "c", "b", "a"))
  stratified_held <- list(c(1L, 3L, 4L, 6L), c(2L, 5L, 7L, 8L), 9:11)
  stratified <- rsample::manual_rset(lapply(stratified_held, function(h) {
    rsample::make_splits(list(analysis = setdiff(1:11, h), assessment = h), small)
  }), c("First", "Second", "Third"))
  stopifnot(identical(sort(unlist(stratified_held)), 1:11))

  # All eight trainable paths: binary, shared OvO/OvR, and moderated regression,
  # each with an ordered holdout and ordinary non-shuffled Cross folds.
  reference_results <- list()
  for (kind in c("binary", "ovo", "ovr", "regression")) {
    data <- data.frame(x = sin(1:72), z = cos(1:72))
    data$truth <- if (kind == "regression") 10 + 2 * data$x + data$z else
      factor(rep(if (kind == "binary") c("working", "fault") else c("soil", "road", "water"),
                 length.out = nrow(data)))
    # Final assessment is excluded from model learning AND calibration.
    development <- data[1:60, ]
    assessment <- data[61:72, ]
    spec <- switch(kind, regression = parsnip::linear_reg(engine = "lm"),
                   binary = parsnip::logistic_reg(engine = "glm"),
                   parsnip::multinom_reg(engine = "nnet", penalty = .1))
    base <- workflows::workflow() |>
      workflows::add_recipe(recipes::recipe(truth ~ ., development) |>
                              recipes::step_normalize(recipes::all_numeric_predictors())) |>
      workflows::add_model(spec)
    # These are literal local positions, not forward validation. In particular,
    # the first Cross model trains on later rows 21:60.
    cross <- rsample::manual_rset(lapply(list(1:20, 21:40, 41:60), function(h) {
      rsample::make_splits(list(analysis = setdiff(1:60, h), assessment = h), development)
    }), c("Early", "Middle", "Late"))
    for (method in c("inductive", "cross")) {
      plan <- if (method == "inductive") ordered_holdout(development) else cross
      resolve <- local_calibration_splits(list(plan), names(development))
      options <- if (kind == "regression") list(m = 1L) else list(classification = kind)
      model <- do.call(calibrated_workflow, c(list(workflow = base, outcome = "truth",
        method = method, splits = resolve, seed = 42), options))
      fit <- generics::fit(model, development)
      p <- predict(fit, assessment)
      score <- if (kind == "regression") {
        c(rmse = sqrt(mean((p$.pred - assessment$truth)^2)))
      } else {
        p <- predict(fit, assessment, type = "prob")
        probability <- as.matrix(p)
        colnames(probability) <- sub("^\\.pred_", "", colnames(probability))
        c(brier = brier_score(assessment$truth, probability),
          log_loss = log_loss(assessment$truth, probability))
      }
      state <- workflows::extract_fit_engine(fit)
      reference_results[[paste(kind, method, sep = "_")]] <- list(
        model = model, fit = fit, development = development, assessment = assessment,
        prediction = p, score = score, folds = state$folds)
      stopifnot(all(is.finite(score)), identical(names(state$folds), names(state$models)))
    }
  }
  # For mixed plans, use splits = NULL with seed, or a seeded initial_split()/
  # vfold_cv() callback. Automatic classification stratifies by outcome unless
  # strata = FALSE; regression does not stratify. Same seed requires the same
  # R RNGkind, package/engine versions and model settings for reproducibility.
  # R/Python seeds alone do not prove equal assignments or model outputs.
  #
  # Machine-failure detection is classification, not a failure-time forecast.
  # Temperature forecast correction is regression; its subject does not make
  # these partitions time-series validation. No rolling window, horizon, embargo,
  # grouping or online update is introduced. Cross uses exact fold complements.
  # Scores may get worse. VA bounds are not individual confidence/prediction
  # intervals; neither dependent data nor Cross aggregates gain universal coverage.
}

Independent Inductive training and calibration sizes

inductive-sizes.R: Request 50 training and 20 calibration rows out of 100, leaving 30 unassigned rows that only a full-model refit learns from, then resolve the same sizes on every class pair of a pairwise multiclass fit.

# Independent Inductive training and calibration sizes, as in sklearn's
# train_size/test_size. Rows in neither set form the unassigned set U: the
# matched source model and the Venn-Abers calibrator never see them, while a
# full-model refit (Full1) learns on all local rows, including C and U.
library(vennabers)
source(system.file("examples", "assertions.R", package = "vennabers"))

if (all(vapply(c("parsnip", "workflows", "rsample", "withr"),
               requireNamespace, logical(1), quietly = TRUE))) {
  data <- data.frame(x = sin(1:120) + rep(c(1, 0, 0, 1), 30),
                     truth = factor(rep(c("yes", "no", "no", "yes"), 30)))
  development <- data[1:100, ]
  assessment <- data[101:120, ]
  base <- workflows::workflow() |>
    workflows::add_formula(truth ~ x) |>
    workflows::add_model(parsnip::logistic_reg(engine = "glm"))

  # 100 local rows: 50 train the source model, 20 calibrate, 30 stay unassigned.
  # Whole numbers are exact row counts; a value in (0, 1) is a proportion
  # (training rounds down, calibration rounds up). Classification draws are
  # stratified by outcome with exact totals unless strata = FALSE.
  fits <- list()
  for (mode in c("matched", "full")) {
    fits[[mode]] <- generics::fit(calibrated_workflow(base, "truth", seed = 42,
      training_size = 50, calibration_size = 20, prediction_model = mode), development)
    state <- workflows::extract_fit_engine(fits[[mode]])
    print(state)
    roles <- state$folds$Inductive1
    assert_conditions(length(roles$analysis) == 50L, length(roles$assessment) == 20L,
      length(roles$unassigned) == 30L, identical(state$calibration$Inductive1$row, roles$assessment))
    stopifnot(identical(sort(c(roles$analysis, roles$assessment, roles$unassigned)), 1:100),
      identical(unname(state$sources), if (mode == "full") "Full1" else "Inductive1"))
  }
  # Both modes calibrate on the same 20 rows; only the prediction model differs.
  stopifnot(identical(workflows::extract_fit_engine(fits$matched)$folds,
                      workflows::extract_fit_engine(fits$full)$folds))
  for (mode in names(fits)) {
    p <- predict(fits[[mode]], assessment, type = "prob")
    probability <- as.matrix(p)
    colnames(probability) <- sub("^\\.pred_", "", colnames(probability))
    print(c(mode = mode, brier = round(brier_score(assessment$truth, probability), 4)))
  }

  # A single size takes the complement, so U is empty: 70 training rows, 30 C.
  complement <- workflows::extract_fit_engine(generics::fit(calibrated_workflow(base, "truth",
    seed = 42, training_size = .7), development))
  stopifnot(lengths(complement$folds$Inductive1[c("analysis", "assessment", "unassigned")]) == c(70L, 30L, 0L))

  # Sizes are checked strictly against the local rows and never shrunk.
  message <- tryCatch(generics::fit(calibrated_workflow(base, "truth", seed = 42,
    training_size = 80, calibration_size = 30), development), error = conditionMessage)
  stopifnot(grepl("request 110 rows, but only the 100 local rows are available", message, fixed = TRUE))

  # Pairwise multiclass resolves the sizes on each class pair's rows, with U
  # per pair. Classes of 40, 30 and 20 rows give pairs of 70, 60 and 50 rows:
  # half of each pair trains, 10 rows calibrate and the rest stay unassigned.
  three <- data.frame(x = sin(1:90) + rep(c(0, .5, 1), c(40, 30, 20)),
                      truth = factor(rep(c("a", "b", "c"), c(40, 30, 20))))
  pairwise <- workflows::extract_fit_engine(generics::fit(calibrated_workflow(base, "truth",
    classification = "ovo", model_training = "pairwise", seed = 42,
    training_size = .5, calibration_size = 10), three))
  print(pairwise)
  stopifnot(identical(pairwise$parameters$effective$unassigned_size,
                      c(pair_1 = 25L, pair_2 = 20L, pair_3 = 15L)))
  # Searches whose candidates are calibrated workflows (calibration or joint
  # model tuning) resolve the sizes again on resample rows and therefore
  # accept proportions only; two-stage model searches also take row counts.
}

Own fit and predict functions

model-adapter.R: Wrap GLM, multinomial and linear regression fit/predict functions with model_adapter(), calibrate them like native engines and keep their settings bound for saved fits. The GLM callback also carries pairwise OvO, Sigmoid postprocessing and a grid search over a recipe degree and the VA loss.

# model_adapter(): your own fit(x, y) and predict(object, new_data) functions in
# a calibrated workflow, without registering a parsnip engine. The workflow keeps
# preprocessing, splits and calibration: every local fit trains a fresh model on
# its own prepared rows, and the calibrator never sees those training rows.
library(vennabers)
source(system.file("examples", "assertions.R", package = "vennabers"))

examples <- list()
if (all(vapply(c("parsnip", "workflows", "recipes", "rsample", "withr", "nnet"),
               requireNamespace, logical(1), quietly = TRUE))) {
  # Each factory binds its helpers and fixed settings locally, so a saved fit
  # carries the actual functions into a fresh R process. Package functions use
  # pkg::fun; nothing is looked up in the global environment later.
  glm_adapter <- function() model_adapter("classification",
    fit = function(x, y) list(model = stats::glm(y ~ ., data = cbind(x, y = y), family = stats::binomial()),
                              classes = levels(y)),
    predict = function(object, new_data) {
      # A binomial GLM models the second level. Return both named columns.
      p <- stats::predict(object$model, new_data, type = "response")
      matrix(c(1 - p, p), ncol = 2L, dimnames = list(NULL, object$classes))
    })
  multinom_adapter <- function(decay = 0) {
    force(decay)  # bind the value now, not a promise to the caller's variable
    model_adapter("classification",
      fit = function(x, y) nnet::multinom(y ~ ., data = cbind(x, y = y), decay = decay, trace = FALSE),
      predict = function(object, new_data) {
        # multinom drops a single row to a vector; keep one column per class.
        p <- stats::predict(object, new_data, type = "probs")
        matrix(p, nrow(new_data), length(object$lev), dimnames = list(NULL, object$lev))
      })
  }
  lm_adapter <- function() model_adapter("regression",
    fit = function(x, y) stats::lm(y ~ ., data = cbind(x, y = y)),

  explicit <- generics::fit(calibrated_workflow(base, schema$encoded_outcome, schema = schema,
    model_training = "pairwise", splits = plans, seed = 42), development)
  stopifnot(identical(native, venn_abers_details(explicit, assessment)))
  # Independent complete binary workflows: preprocessing sees only each pair's
  # training rows, and held-out model probabilities fit the existing binary core.
  q <- matrix(0, nrow(assessment), nrow(pair_table))
  for (j in seq_along(plans)) {
    training <- rsample::analysis(plans[[j]])
    held <- rsample::assessment(plans[[j]])
    reference <- workflows::workflow() |> workflows::add_recipe(
      recipes::recipe(formula, training) |> recipes::step_normalize(recipes::all_numeric_predictors())) |>
      workflows::add_model(parsnip::logistic_reg(engine = "glm"))
    binary <- generics::fit(reference, training)
    event <- pair_table$class_2[j]
    column <- paste0(".pred_", event)
    cal <- venn_abers(held[[schema$encoded_outcome]] == event, predict(binary, held, type = "prob")[[column]])
    expected <- predict(cal, predict(binary, assessment, type = "prob")[[column]])
    parts <- native$details[native$details$pair_id == pair_table$pair_id[j], c("lower", "upper", "probability")]
    stopifnot(all(abs(as.matrix(parts) - as.matrix(expected)) <= 1e-10))
    q[, j] <- expected$probability
  }
  # Reciprocal OvO coupling, written independently as sums of binary odds.
  p <- vapply(state$class_levels, function(cl) {
    relevant <- which(pair_table$class_1 == cl | pair_table$class_2 == cl)
    odds <- lapply(relevant, function(j) {
      probability <- if (pair_table$class_2[j] == cl) q[, j] else 1 - q[, j]
      (1 - probability) / probability
    })
    1 / (1 + Reduce(`+`, odds))
  }, numeric(nrow(assessment)))
  stopifnot(all(abs(p / rowSums(p) - as.matrix(native$prediction)) <= 1e-10))
  if (all(vapply(c("tune", "yardstick"), requireNamespace, logical(1), quietly = TRUE))) {
    scoring <- yardstick::metric_set(yardstick::brier_class, yardstick::mn_log_loss)
    outer <- withr::with_seed(41, rsample::vfold_cv(development, v = 2, repeats = 2))
    result <- tune::fit_resamples(model, outer, metrics = scoring,
      control = tune::control_resamples(save_pred = TRUE, extract = identity, allow_par = FALSE))
    checker <- new.env()
    sys.source(system.file("examples", "check-encoded-evaluation.R", package = "vennabers"), checker)
    checked <- checker$check_encoded_evaluation(result, outer, schema)
    print(checked$metrics)
    final <- tune::last_fit(model, outer_split, metrics = scoring)
    predictions <- tune::collect_predictions(final)
    metrics <- tune::collect_metrics(final)
    assert_conditions(nrow(tune::collect_notes(final)) == 0L, identical(predictions$.row, 91:120),
      !anyDuplicated(predictions$.row), all(predictions$.config == "pre0_mod0_post0"),
      identical(predictions[[schema$encoded_outcome]], assessment[[schema$encoded_outcome]]),
      nrow(metrics) == 2L, all(is.finite(metrics$.estimate)),
      all(abs(as.matrix(predictions[names(native$prediction)]) - as.matrix(native$prediction)) <= 1e-10))
  }
  # Matched: P = K*(K-1)/2 pairs cost P base fits and P predictions per nonempty
  # batch. No extra shared model or OvR ensemble is trained. All P recipes and
  # calibrators remain available; predict never learns. A row may belong to
  # several pairs.
  # Outer assessment labels never reach these local fits or calibration plans.
  # Prepared outer plans can use local_calibration_splits() over all local pairs;
  # a development-wide plan cannot be reused on a smaller outer analysis domain.
  # Matching-model assumptions still apply; no universal accuracy gain is promised.
  # Full transfer keeps the held-out calibrators but fits one extra workflow
  # on each complete local pair, including its calibration rows.
  full_model <- calibrated_workflow(base, schema$encoded_outcome, schema = schema,
    model_training = "pairwise", method = "inductive", prediction_model = "full",
    splits = pair_plan, seed = 42)
  full_fit <- generics::fit(full_model, development)
  full_state <- workflows::extract_fit_engine(full_fit)
  for (id in names(state$pair_models)) stopifnot(
    identical(state$pair_models[[id]]$calibration, full_state$pair_models[[id]]$calibration),
    identical(state$pair_models[[id]]$calibrator, full_state$pair_models[[id]]$calibrator),
    identical(names(full_state$pair_models[[id]]$models), "Full1"))
  full_native <- venn_abers_details(full_fit, assessment)
  full_decoded <- venn_abers_decode(full_native, schema)
  print(head(full_decoded$details))
  for (mode in c("matched", "full")) {
    probability <- if (mode == "full") full_decoded$probability else decoded$probability
    truth <- venn_abers_decode(assessment[schema$encoded_outcome], schema)$truth
    cat(mode, "Brier:", brier_score(truth, probability),
        "log loss:", log_loss(truth, probability), "\n")
  }
  # Six base fits (three sources, three Full1); only the Full1 workflows remain,
  # each predicting once per nonempty batch. The printed slot map sends every
  # held-out calibrator to its pair's Full1.
  print(full_state)
  # Each pair source is named pair_N/Full1; pair_id + fold identifies its
  # calibrator. Source workflows are discarded after all full fits succeed.
  # No in-sample recalibration or pooling occurs. Outer assessment stays excluded.
  # Matching-model guarantees do not automatically transfer to the new model;
  # neither coverage nor improved quality is promised. Settings are fixed here;
  # pairwise-grid.R and pairwise-bayes.R show automatic local selection.
}
